const formatETB = (amount) => {
  const num = Number(amount) || 0;
  return `${num.toLocaleString('en-US', {
    minimumFractionDigits: num % 1 === 0 ? 0 : 2,
    maximumFractionDigits: 2,
  })} ETB`;
};

const generateTransactionRef = (method) => {
  const prefix =
    method === 'TELEBIRR' ? 'TLB' : method === 'CBE_BIRR' ? 'CBE' : 'COD';
  const rand = Math.random().toString(36).substring(2, 8).toUpperCase();
  const stamp = Date.now().toString().slice(-4);
  return `${prefix}-${stamp}-${rand}`;
};

const generateSecureQrToken = () => {
  const part1 = Math.random().toString(36).substring(2, 7).toUpperCase();
  const part2 = Math.random().toString(36).substring(2, 7).toUpperCase();
  return `CHUKO-QR-${part1}-${part2}`;
};

// Deterministic Visual SVG QR Matrix Generator (renders crisp unique QR graphic)
const SecureQrSvg = ({ token, size = 148 }) => {
  const grid = useMemo(() => {
    const cells = [];
    let hash = 0;
    for (let i = 0; i < token.length; i++) {
      hash = (hash << 5) - hash + token.charCodeAt(i);
      hash |= 0;
    }
    for (let r = 0; r < 15; r++) {
      const row = [];
      for (let c = 0; c < 15; c++) {
        // Finder patterns in 3 corners
        const isTopLeft = r < 4 && c < 4;
        const isTopRight = r < 4 && c > 10;
        const isBottomLeft = r > 10 && c < 4;
        if (isTopLeft || isTopRight || isBottomLeft) {
          const border =
            r === 0 ||
            r === 3 ||
            c === 0 ||
            c === 3 ||
            r === 11 ||
            r === 14 ||
            c === 11 ||
            c === 14;
          const center =
            (r >= 1 && r <= 2 && c >= 1 && c <= 2) ||
            (r >= 1 && r <= 2 && c >= 12 && c <= 13) ||
            (r >= 12 && r <= 13 && c >= 1 && c <= 2);
          row.push(border || center ? 1 : 0);
        } else {
          const val = Math.abs(Math.sin(hash + r * 17 + c * 31));
          row.push(val > 0.48 ? 1 : 0);
        }
      }
      cells.push(row);
    }
    return cells;
  }, [token]);

  return (
    <div className="inline-flex flex-col items-center p-3 bg-white rounded-2xl border-2 border-[#3A1E10]/20 shadow-sm">
      <svg
        width={size}
        height={size}
        viewBox="0 0 15 15"
        shapeRendering="crispEdges"
      >
        <rect width="15" height="15" fill="#FFFFFF" />
        {grid.map((row, rIdx) =>
          row.map((cell, cIdx) =>
            cell ? (
              <rect
                key={`${rIdx}-${cIdx}`}
                x={cIdx}
                y={rIdx}
                width="1"
                height="1"
                fill="#2B1508"
              />
            ) : null
          )
        )}
      </svg>
      <span className="mt-2 font-mono text-xs font-bold tracking-wider text-[#3A1E10] bg-[#FDF8F2] px-2.5 py-1 rounded border border-[#F39223]/40">
        {token}
      </span>
    </div>
  );
};

      triggerToast(`Order #${order.orderNumber} marked ${newStatus}`);

      console.error('Payment verify error:', err);

    if (!user) return;
    try {
      await updateDoc(
        doc(db, 'artifacts', appId, 'public', 'data', 'orders', order.id),
        {
          assignedDriver: driverProfile,
          orderStatus: 'ASSIGNED',
          driverAccepted: false,
        }
      );
      await pushSystemEvent(
        'Driver Assigned',
        `${driverProfile.name} assigned to Order #${order.orderNumber}`,
        'ADMIN'
      );
      triggerToast(`Assigned ${driverProfile.name} to Order #${order.orderNumber}`);
    } catch (err) {
      console.error('Assign driver error:', err);
    }

  // CUSTOMER ACTIONS: CART, ORDER PLACEMENT, QR DISPLAY, REVIEWS
  // ==========================================================================
  const handleAddToCart = (food) => {
    const qty = parseFloat(qtySelections[food.id] || 1);
    if (isNaN(qty) || qty <= 0) return;

    setCart((prev) => {
      const existing = prev.find((i) => i.foodId === food.id);
      if (existing) {
        return prev.map((i) =>
          i.foodId === food.id
            ? { ...i, quantityKg: Number((i.quantityKg + qty).toFixed(2)) }
            : i
        );
      }
      return [
        ...prev,
        {
          foodId: food.id,
          name: food.name,
          pricePerKg: food.pricePerKg,
          quantityKg: qty,
          imageUrl: food.imageUrl,
        },
      ];
    });
    triggerToast(`${qty} kg of ${food.name} added to cart.`);
  };

  const handlePlaceOrder = async (e) => {
    e.preventDefault();
    if (!user || cart.length === 0) return;

    const orderNumber = `AC-${Math.floor(100000 + Math.random() * 900000)}`;
    const txRef =
      customerTxRef.trim() || generateTransactionRef(paymentMethod);
    const qrToken = generateSecureQrToken();

    // Per security rule: Never auto-mark Telebirr/CBE Paid based on customer input alone;
    // Cash on Delivery starts UNPAID, Mobile Money starts PENDING until server/admin verification.
    const initialPaymentStatus =
      paymentMethod === 'CASH_ON_DELIVERY' ? 'UNPAID' : 'PENDING';

    const newOrder = {
      orderNumber,
      customerId: user.uid,
      customerName: customerName.trim() || 'Valued Guest',
      customerPhone: customerPhone.trim() || '+251 900 000 000',
      items: cart,
      fulfillmentType,
      route:
        fulfillmentType === 'DELIVERY'
          ? activeRoute
          : {
              id: 'pickup',
              name: 'Arsi Chuko Flagship Kitchen (Self Pickup)',
              distanceKm: 0,
              etaMins: 0,
            },
      perKmRateSnapshot: pricingConfig.perKmRate,
      foodSubtotal: cartTotals.foodSubtotal,
      serviceCharge: cartTotals.serviceCharge,
      distanceKm: cartTotals.distanceKm,
      transportationFee: cartTotals.transportationFee,
      grandTotal: cartTotals.grandTotal,
      deliveryDate,
      deliveryTime,
      specialInstructions: specialInstructions.trim(),
      paymentMethod,
      paymentStatus: initialPaymentStatus,
      txRef,
      qrToken,
      qrVerified: false,
      qrVerifiedAt: null,
      orderStatus: 'PENDING',
      assignedDriver: null,
      driverAccepted: false,
      gpsSharing: false,
      createdAt: Date.now(),
    };

    try {
      const docRef = await addDoc(
        collection(db, 'artifacts', appId, 'public', 'data', 'orders'),
        newOrder
      );
      await pushSystemEvent(
        'New Order Placed',
        `Order #${orderNumber} (${formatETB(newOrder.grandTotal)}) via ${paymentMethod}`,
        'CUSTOMER'
      );
      setCart([]);
      setCustomerTxRef('');
      setShowCartDrawer(false);
      setCustomerTab('orders');
      setReceiptOrder({ id: docRef.id, ...newOrder });
      triggerToast(`Order #${orderNumber} placed! Unique QR & Receipt generated.`);
    } catch (err) {
      console.error('Place order error:', err);
      triggerToast('Failed to place order.', 'error');
    }
  };

  const handleSubmitReview = async (e) => {
    e.preventDefault();
    if (!user || !reviewOrderTarget) return;
    try {
      await addDoc(
        collection(db, 'artifacts', appId, 'public', 'data', 'reviews'),
        {
          orderId: reviewOrderTarget.id,
          orderNumber: reviewOrderTarget.orderNumber,
          customerName: reviewOrderTarget.customerName,
          foodRating: Number(reviewForm.foodRating),
          deliveryRating: Number(reviewForm.deliveryRating),
          comment: reviewForm.comment.trim(),
          hiddenByAdmin: false,
          createdAt: Date.now(),
        }

      await updateDoc(
        doc(
          db,
          'artifacts',
          appId,
          'public',
          'data',
          'orders',
          reviewOrderTarget.id
        ),
        { hasReviewed: true }
      );
      await pushSystemEvent(
        'Customer Review Submitted',
        `Order #${reviewOrderTarget.orderNumber} rated ${reviewForm.foodRating}/5 stars`,
        'CUSTOMER'
      );
      setReviewOrderTarget(null);
      setReviewForm({ foodRating: 5, deliveryRating: 5, comment: '' });
      triggerToast('Thank you! Your verified review has been recorded.');
    } catch (err) {
      console.error('Review submit error:', err);
    }
  };

  // DRIVER ACTIONS: ACCEPT/REJECT, LIVE GPS, CASH CONFIRM, ONE-TIME QR SCAN
  // ==========================================================================
  const handleDriverAcceptReject = async (order, accept) => {
    if (!user) return;
    try {
      if (accept) {
        await updateDoc(
          doc(db, 'artifacts', appId, 'public', 'data', 'orders', order.id),
          {
            driverAccepted: true,
            orderStatus: 'IN_TRANSIT',
            gpsSharing: true,

          'DRIVER'

        triggerToast(`Accepted Order #${order.orderNumber}. Live GPS active.`);
      } else {
        await updateDoc(
          doc(db, 'artifacts', appId, 'public', 'data', 'orders', order.id),
          {
            assignedDriver: null,
            driverAccepted: false,
            orderStatus: 'CONFIRMED',
            gpsSharing: false,
          }
        );
        await pushSystemEvent(
          'Driver Rejected Assignment',
          `${driverProfile.name} declined Order #${order.orderNumber}`,
          'DRIVER'

      console.error('Driver accept/reject error:', err);

    const nextState = !order.gpsSharing;
    try {
      await updateDoc(
        doc(db, 'artifacts', appId, 'public', 'data', 'orders', order.id),
        { gpsSharing: nextState }
      );
      setGpsSharingOrderId(nextState ? order.id : null);
      triggerToast(
        nextState ? 'Live GPS location sharing started' : 'GPS sharing paused'
      );

      console.error('GPS toggle error:', err);
    }

        {
          driverCashCollected: true,

        'Driver Reported Cash Collection',
        `${driverProfile.name} submitted cash collection for Order #${order.orderNumber} (Awaiting Admin Verification)`,
        'DRIVER'

        'Cash collection logged. Pending Admin verification.'
      );

      console.error('Cash confirm error:', err);

  const handleVerifyQrToken = async (e) => {
    e.preventDefault();
    if (!user) return;
    const cleanToken = qrInputToken.trim().toUpperCase();
    if (!cleanToken) return;

    const matchedOrder = orders.find(
      (o) => (o.qrToken || '').toUpperCase() === cleanToken
    );

    if (!matchedOrder) {
      setQrScanResult({
        status: 'INVALID',
        message: 'Token does not match any active or confirmed order.',
        order: null,
      });
      return;
    }

    if (matchedOrder.qrVerified) {
      setQrScanResult({
        status: 'ALREADY_USED',
        message: `This one-time QR token was already redeemed on ${new Date(
          matchedOrder.qrVerifiedAt
        ).toLocaleTimeString()}.`,
        order: matchedOrder,
      });
      return;
    }

    // Valid one-time verification (Kept strictly separate from payment status)
    try {
      await updateDoc(
        doc(
          db,
          'artifacts',
          appId,
          'public',
          'data',
          'orders',
          matchedOrder.id
        ),
        {
          qrVerified: true,
          qrVerifiedAt: Date.now(),
          orderStatus: 'DELIVERED',
          gpsSharing: false,
        }
      );
      if (gpsSharingOrderId === matchedOrder.id) {
        setGpsSharingOrderId(null);
      }
      await pushSystemEvent(
        'QR Code Verified & Delivered',
        `Order #${matchedOrder.orderNumber} QR redeemed (${cleanToken})`,
        'DRIVER'
      );
      setQrScanResult({
        status: 'VERIFIED',
        message: `Delivery verified for Order #${matchedOrder.orderNumber} (${matchedOrder.customerName}). Payment status remains ${matchedOrder.paymentStatus}.`,
        order: matchedOrder,
      });
      setQrInputToken('');
    } catch (err) {
      console.error('QR verify error:', err);
    }

  const analytics = useMemo(() => {
    const paidOrders = orders.filter((o) => o.paymentStatus === 'PAID');
    const unpaidOrders = orders.filter((o) => o.paymentStatus === 'UNPAID');
    const pendingOrders = orders.filter((o) => o.paymentStatus === 'PENDING');
    const completedOrders = orders.filter((o) => o.orderStatus === 'DELIVERED');
    const cancelledOrders = orders.filter((o) => o.orderStatus === 'CANCELLED');

    const totalSales = paidOrders.reduce(
      (sum, o) => sum + (o.grandTotal || 0),
      0
    );
    const deliveryFeesCollected = paidOrders.reduce(
      (sum, o) => sum + (o.transportationFee || 0),
      0
    );
    const serviceChargesCollected = paidOrders.reduce(
      (sum, o) => sum + (o.serviceCharge || 0),
      0
    );

    // Popular foods count
    const foodMap = {};
    orders.forEach((o) => {
      (o.items || []).forEach((item) => {
        if (!foodMap[item.name]) {
          foodMap[item.name] = { name: item.name, kgSold: 0, revenue: 0 };
        }
        foodMap[item.name].kgSold += item.quantityKg || 0;
        foodMap[item.name].revenue +=
          (item.pricePerKg || 0) * (item.quantityKg || 0);
      });
    });
    const popularFoods = Object.values(foodMap).sort(
      (a, b) => b.kgSold - a.kgSold
    );

    const avgFoodRating =
      reviews.length > 0
        ? (
            reviews.reduce((s, r) => s + (r.foodRating || 5), 0) / reviews.length
          ).toFixed(1)
        : 'N/A';

    return {
      totalSales,
      paidCount: paidOrders.length,
      unpaidCount: unpaidOrders.length,
      pendingCount: pendingOrders.length,
      completedCount: completedOrders.length,
      cancelledCount: cancelledOrders.length,
      deliveryFeesCollected,
      serviceChargesCollected,
      popularFoods,
      avgFoodRating,
    };
  }, [orders, reviews]);

      'ServiceCharge_ETB',

      'TxRef',

      o.fulfillmentType,
      o.distanceKm,
      o.foodSubtotal,
      o.serviceCharge,
      o.transportationFee,
      o.grandTotal,
      o.paymentMethod,
      o.paymentStatus,
      o.txRef,
      o.orderStatus,
      o.qrVerified ? 'YES' : 'NO',
    ]);
    const csvContent =
      'data:text/csv;charset=utf-8,' +
      [headers.join(','), ...rows.map((e) => e.join(','))].join('\n');
    const encodedUri = encodeURI(csvContent);
    const link = document.createElement('a');
    link.setAttribute('href', encodedUri);
    link.setAttribute(
      'download',
      `Arsi_Chuko_Financial_Report_${new Date().toISOString().slice(0, 10)}.csv`

    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    triggerToast('Financial CSV report exported.');
  };

  // RENDER
  // ==========================================================================
  return (

      {/* TOAST BANNER */}
      {toast && (
        <div
          className={`fixed bottom-5 right-5 z-50 flex items-center gap-3 px-4 py-3 rounded-xl shadow-xl border text-sm font-medium transition-all ${
            toast.type === 'error'
              ? 'bg-rose-900 text-white border-rose-700'
              : 'bg-[#2B1508] text-[#FDF8F2] border-[#F39223]'
          }`}
        >
          <Sparkles className="w-4 h-4 text-[#F39223] shrink-0" />
          <span>{toast.message}</span>
        </div>
      )}

      {/* ====================================================================
          TOP BRAND HEADER & ROLE / LANGUAGE SWITCHER
      ==================================================================== */}
      <header className="sticky top-0 z-40 bg-[#2B1508] text-[#FDF8F2] border-b-2 border-[#F39223] shadow-lg">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 py-3 flex flex-wrap items-center justify-between gap-3">
          {/* Brand Logo & Title */}
          <div className="flex items-center gap-3">
            <div className="bg-black/40 p-1.5 rounded-2xl border border-[#F39223]/40 shadow-inner">
              <ArsiChukoLogo size={38} />
            </div>
            <div>
              <div className="flex items-center gap-2">
                <h1 className="text-base sm:text-lg font-extrabold tracking-wide text-[#F39223]">
                  {t.brandName}
                </h1>
                <span className="hidden md:inline-block text-[10px] font-bold uppercase tracking-wider px-2 py-0.5 rounded bg-[#F39223]/20 text-[#FFA73B] border border-[#F39223]/40">
                  {t.currency} • {t.timezoneLabel}
                </span>
              </div>
              <p className="text-xs text-[#E6D5C3] hidden sm:block">
                {t.brandTagline}
              </p>
            </div>
          </div>

          {/* Right Controls: Language Selector, Role Switcher, Cart & Notifications */}
          <div className="flex flex-wrap items-center gap-2 sm:gap-3">
            {/* Three-Language Switcher (EN / OM / AM) */}
            <div className="flex items-center bg-[#3A1E10] border border-[#F39223]/40 rounded-xl p-1">
              <Globe className="w-3.5 h-3.5 text-[#F39223] ml-1.5 mr-1" />
              {[
                { code: 'en', label: 'EN' },
                { code: 'om', label: 'Afaan Oromoo' },
                { code: 'am', label: 'አማርኛ' },
              ].map((l) => (
                <button
                  key={l.code}

              title={t.notificationsTitle}
            >
              <Bell className="w-4 h-4 text-[#F39223]" />
              {notifications.length > 0 && (
                <span className="absolute -top-1 -right-1 w-4 h-4 rounded-full bg-[#F39223] text-[#2B1508] text-[10px] font-extrabold flex items-center justify-center">
                  {notifications.length > 9 ? '9+' : notifications.length}
                </span>
              )}
            </button>

            {/* Customer Cart Button */}
            {role === 'CUSTOMER' && (
              <button
                onClick={() => setShowCartDrawer(true)}
                className="flex items-center gap-2 px-3.5 py-2 rounded-xl bg-[#F39223] hover:bg-[#FFA73B] text-[#2B1508] font-extrabold text-xs sm:text-sm shadow-md transition"
              >
                <ShoppingBag className="w-4 h-4" />
                <span>{t.navCart}</span>
                <span className="px-1.5 py-0.5 rounded-full bg-[#2B1508] text-[#F39223] text-xs">
                  {cart.length}

                    customerTab === 'orders'

                      </button>
                      <button
                        type="button"
                        onClick={() => setFulfillmentType('PICKUP')}
                        className={`py-2.5 px-3 rounded-xl text-xs font-bold border transition ${
                          fulfillmentType === 'PICKUP'
                            ? 'bg-[#2B1508] text-[#F39223] border-[#2B1508]'
                            : 'bg-white text-[#3A1E10] border-[#3A1E10]/20'
                        }`}
                      >
                        {t.pickupOption}
                      </button>
                    </div>
                  </div>

                  {/* Real Road Route Selector (Only when Delivery selected) */}
                  {fulfillmentType === 'DELIVERY' && (
                    <div className="bg-white p-3.5 rounded-xl border border-[#F39223]/50 space-y-2">
                      <label className="block text-xs font-bold text-[#2B1508]">
                        {t.deliveryZone}
                      </label>
                      <select
                        value={selectedRouteId}
                        onChange={(e) => setSelectedRouteId(e.target.value)}
                        className="w-full px-3 py-2 rounded-xl border border-[#3A1E10]/20 text-xs font-semibold bg-[#FDF8F2]"
                      >
                        {DELIVERY_ROUTES.map((r) => (
                          <option key={r.id} value={r.id}>
                            {r.name} — {r.distanceKm} km (~{r.etaMins} mins)
                          </option>
                        ))}
                      </select>
                      <div className="flex items-center justify-between text-xs text-[#693E1E] pt-1">
                        <span>
                          {t.roadDistance}:{' '}
                          <strong>{activeRoute.distanceKm} km</strong>
                        </span>
                        <span>
                          Rate: <strong>{pricingConfig.perKmRate} ETB/km</strong>
                        </span>
                      </div>
                    </div>
                  )}

                  {/* Delivery Date & Exact Time + Special Instructions */}
                  <div className="grid grid-cols-2 gap-3">
                    <div>
                      <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                        {t.deliveryDate}
                      </label>
                      <input
                        type="date"
                        required
                        value={deliveryDate}
                        onChange={(e) => setDeliveryDate(e.target.value)}
                        className="w-full px-3 py-2 rounded-xl border border-[#3A1E10]/20 bg-white text-xs font-semibold"
                      />
                    </div>
                    <div>
                      <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                        {t.deliveryTime}
                      </label>
                      <input
                        type="time"
                        required
                        value={deliveryTime}
                        onChange={(e) => setDeliveryTime(e.target.value)}
                        className="w-full px-3 py-2 rounded-xl border border-[#3A1E10]/20 bg-white text-xs font-semibold"
                      />
                    </div>
                  </div>

                  <div>
                    <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                      {t.specialInstructions}
                    </label>
                    <textarea
                      rows="2"
                      value={specialInstructions}
                      onChange={(e) => setSpecialInstructions(e.target.value)}
                      placeholder={t.instructionsPlaceholder}
                      className="w-full px-3 py-2 rounded-xl border border-[#3A1E10]/20 bg-white text-xs"
                    />
                  </div>

                  {/* Payment Method Selection */}
                  <div className="space-y-2">
                    <label className="block text-xs font-bold text-[#3A1E10]">
                      {t.paymentMethod}
                    </label>
                    <div className="grid grid-cols-3 gap-2">
                      {[
                        { id: 'TELEBIRR', label: t.payTelebirr },
                        { id: 'CBE_BIRR', label: t.payCbe },
                        { id: 'CASH_ON_DELIVERY', label: t.payCod },
                      ].map((pm) => (
                        <button
                          key={pm.id}
                          type="button"
                          onClick={() => setPaymentMethod(pm.id)}
                          className={`p-2 rounded-xl text-[11px] font-bold border text-center transition ${
                            paymentMethod === pm.id
                              ? 'bg-[#F39223] text-[#2B1508] border-[#2B1508]'
                              : 'bg-white text-[#3A1E10] border-[#3A1E10]/20'
                          }`}
                        >
                          {pm.label}
                        </button>
                      ))}
                    </div>

                    {paymentMethod !== 'CASH_ON_DELIVERY' && (
                      <input
                        type="text"
                        value={customerTxRef}
                        onChange={(e) => setCustomerTxRef(e.target.value)}
                        placeholder={t.txRefPlaceholder}
                        className="w-full px-3 py-2 rounded-xl border border-[#3A1E10]/20 bg-white text-xs font-mono"
                      />
                    )}
                  </div>

                  {/* Transparent ETB Calculation Summary */}
                  <div className="bg-[#2B1508] text-[#FDF8F2] p-4 rounded-2xl space-y-1.5 text-xs">
                    <h4 className="font-extrabold text-[#F39223] uppercase tracking-wider mb-2">
                      {t.pricingBreakdown}
                    </h4>
                    <div className="flex justify-between">
                      <span>{t.foodSubtotal}:</span>
                      <span className="font-bold">
                        {formatETB(cartTotals.foodSubtotal)}
                      </span>
                    </div>
                    <div className="flex justify-between">
                      <span>{t.serviceCharge}:</span>
                      <span className="font-bold">
                        {formatETB(cartTotals.serviceCharge)}
                      </span>
                    </div>
                    <div className="flex justify-between">
                      <span>
                        {t.transportationFee} ({cartTotals.distanceKm} km ×{' '}
                        {pricingConfig.perKmRate} ETB):
                      </span>
                      <span className="font-bold">
                        {formatETB(cartTotals.transportationFee)}
                      </span>
                    </div>
                    <div className="flex justify-between text-base font-black text-[#F39223] pt-2 border-t border-white/15">
                      <span>{t.grandTotal}:</span>
                      <span>{formatETB(cartTotals.grandTotal)}</span>
                    </div>
                  </div>

                  <button
                    type="submit"
                    className="w-full py-3.5 rounded-xl bg-[#F39223] hover:bg-[#FFA73B] text-[#2B1508] font-extrabold text-sm shadow-lg transition"
                  >
                    {t.placeOrderBtn} • {formatETB(cartTotals.grandTotal)}
                  </button>
                </form>
              )}
            </div>
          </div>
        </div>
      )}

      {/* ====================================================================
          ADMIN ADD / EDIT FOOD MODAL
      ==================================================================== */}
      {showFoodModal && (
        <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4">
          <div className="bg-[#FDF8F2] rounded-3xl max-w-lg w-full border-2 border-[#F39223] shadow-2xl overflow-hidden">
            <div className="p-4 bg-[#2B1508] text-[#FDF8F2] flex items-center justify-between">
              <h3 className="font-extrabold text-base text-[#F39223]">
                {editingFoodId ? t.editFood : t.addNewFood}
              </h3>
              <button
                onClick={() => setShowFoodModal(false)}
                className="p-1 rounded-lg hover:bg-white/10"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveFood} className="p-5 space-y-4">
              <div>
                <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                  {t.foodName} *
                </label>
                <input
                  type="text"
                  required
                  placeholder="e.g., Special Arsi Chuko (Spiced Roasted Barley & Butter)"
                  value={foodForm.name}
                  onChange={(e) =>
                    setFoodForm((p) => ({ ...p, name: e.target.value }))
                  }
                  className="w-full px-3.5 py-2.5 rounded-xl border border-[#3A1E10]/25 bg-white text-sm font-semibold"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                    {t.foodPriceKg} *
                  </label>
                  <input
                    type="number"
                    required
                    min="1"
                    step="0.01"
                    placeholder="500"
                    value={foodForm.pricePerKg}
                    onChange={(e) =>
                      setFoodForm((p) => ({ ...p, pricePerKg: e.target.value }))
                    }
                    className="w-full px-3.5 py-2.5 rounded-xl border border-[#3A1E10]/25 bg-white text-sm font-bold"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                    {t.foodQtyKg} *
                  </label>
                  <input
                    type="number"
                    required
                    min="0"
                    step="0.5"
                    placeholder="25"
                    value={foodForm.availableKg}
                    onChange={(e) =>
                      setFoodForm((p) => ({
                        ...p,
                        availableKg: e.target.value,
                      }))
                    }
                    className="w-full px-3.5 py-2.5 rounded-xl border border-[#3A1E10]/25 bg-white text-sm font-bold"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                  {t.foodCategory}
                </label>
                <input
                  type="text"
                  value={foodForm.category}
                  onChange={(e) =>
                    setFoodForm((p) => ({ ...p, category: e.target.value }))
                  }
                  className="w-full px-3.5 py-2 rounded-xl border border-[#3A1E10]/25 bg-white text-xs font-semibold"
                />
              </div>

              <div>
                <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                  {t.foodImageUrl}
                </label>
                <div className="flex items-center gap-2">
                  <input
                    type="file"
                    accept="image/*"
                    onChange={handleImageUpload}
                    className="text-xs file:mr-2 file:py-1.5 file:px-3 file:rounded-lg file:border-0 file:bg-[#2B1508] file:text-[#F39223] file:font-bold"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                  {t.foodDesc}
                </label>
                <textarea
                  rows="2"
                  value={foodForm.description}
                  onChange={(e) =>
                    setFoodForm((p) => ({ ...p, description: e.target.value }))
                  }
                  className="w-full px-3.5 py-2 rounded-xl border border-[#3A1E10]/25 bg-white text-xs"
                />
              </div>

              <label className="flex items-center gap-2 text-xs font-bold text-[#2B1508]">
                <input
                  type="checkbox"
                  checked={foodForm.isAvailable}
                  onChange={(e) =>
                    setFoodForm((p) => ({
                      ...p,
                      isAvailable: e.target.checked,
                    }))
                  }
                  className="rounded text-[#F39223]"
                />
                <span>{t.availableStatus}</span>
              </label>

              <div className="flex justify-end gap-2 pt-2">
                <button
                  type="button"
                  onClick={() => setShowFoodModal(false)}
                  className="px-4 py-2 rounded-xl border border-[#3A1E10]/20 text-xs font-bold"
                >
                  {t.cancelBtn}
                </button>
                <button
                  type="submit"
                  className="px-5 py-2 rounded-xl bg-[#F39223] text-[#2B1508] text-xs font-extrabold shadow"
                >
                  {t.saveFoodBtn}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* ====================================================================
          PAYMENT RECEIPT & QR VERIFICATION MODAL
      ==================================================================== */}
      {receiptOrder && (
        <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4">
          <div className="bg-white rounded-3xl max-w-md w-full border-2 border-[#F39223] shadow-2xl overflow-hidden">
            <div className="p-4 bg-[#2B1508] text-[#FDF8F2] flex items-center justify-between">
              <div className="flex items-center gap-2">
                <ArsiChukoLogo size={26} />
                <span className="font-extrabold text-sm text-[#F39223]">
                  OFFICIAL ETB RECEIPT
                </span>
              </div>
              <button
                onClick={() => setReceiptOrder(null)}
                className="p-1 rounded-lg hover:bg-white/10"
              >
                <X className="w-5 h-5" />
              </button>
            </div>
            <div className="p-6 space-y-4 text-center">
              <div>
                <p className="font-mono font-black text-lg text-[#2B1508]">
                  #{receiptOrder.orderNumber}
                </p>
                <p className="text-xs text-[#693E1E]">
                  Ref: {receiptOrder.txRef} • {receiptOrder.paymentMethod}
                </p>
                <div className="mt-2">
                  <PaymentBadge status={receiptOrder.paymentStatus} t={t} />
                </div>
              </div>

              <SecureQrSvg token={receiptOrder.qrToken} size={130} />

              <div className="bg-[#FDF8F2] p-3.5 rounded-xl text-xs space-y-1 text-left border border-[#3A1E10]/15">
                <div className="flex justify-between">
                  <span>{t.foodSubtotal}:</span>
                  <strong>{formatETB(receiptOrder.foodSubtotal)}</strong>
                </div>
                <div className="flex justify-between">
                  <span>{t.serviceCharge}:</span>
                  <strong>{formatETB(receiptOrder.serviceCharge)}</strong>
                </div>
                <div className="flex justify-between">
                  <span>
                    {t.transportationFee} ({receiptOrder.distanceKm} km):
                  </span>
                  <strong>{formatETB(receiptOrder.transportationFee)}</strong>
                </div>
                <div className="flex justify-between text-sm font-black text-[#D97706] pt-1.5 border-t border-[#3A1E10]/15">
                  <span>{t.grandTotal}:</span>
                  <span>{formatETB(receiptOrder.grandTotal)}</span>
                </div>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* ====================================================================
          CUSTOMER REVIEW MODAL (POST-DELIVERY)
      ==================================================================== */}
      {reviewOrderTarget && (
        <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4">
          <div className="bg-[#FDF8F2] rounded-3xl max-w-md w-full border-2 border-[#F39223] shadow-2xl overflow-hidden">
            <div className="p-4 bg-[#2B1508] text-[#FDF8F2] flex items-center justify-between">
              <h3 className="font-extrabold text-sm text-[#F39223]">
                {t.leaveReview} (#{reviewOrderTarget.orderNumber})
              </h3>
              <button
                onClick={() => setReviewOrderTarget(null)}
                className="p-1 rounded-lg hover:bg-white/10"
              >
                <X className="w-5 h-5" />
              </button>
            </div>
            <form onSubmit={handleSubmitReview} className="p-5 space-y-4">
              <div>
                <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                  {t.foodRatingLabel}
                </label>
                <select
                  value={reviewForm.foodRating}
                  onChange={(e) =>
                    setReviewForm((p) => ({
                      ...p,
                      foodRating: e.target.value,
                    }))
                  }
                  className="w-full px-3 py-2 rounded-xl border border-[#3A1E10]/20 bg-white text-sm font-bold"
                >
                  {[5, 4, 3, 2, 1].map((n) => (
                    <option key={n} value={n}>
                      {n} ★
                    </option>
                  ))}
                </select>
              </div>
              <div>
                <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                  {t.deliveryRatingLabel}
                </label>
                <select
                  value={reviewForm.deliveryRating}
                  onChange={(e) =>
                    setReviewForm((p) => ({
                      ...p,
                      deliveryRating: e.target.value,
                    }))
                  }
                  className="w-full px-3 py-2 rounded-xl border border-[#3A1E10]/20 bg-white text-sm font-bold"
                >
                  {[5, 4, 3, 2, 1].map((n) => (
                    <option key={n} value={n}>
                      {n} ★
                    </option>
                  ))}
                </select>
              </div>
              <div>
                <label className="block text-xs font-bold text-[#3A1E10] mb-1">
                  {t.reviewCommentLabel}
                </label>
                <textarea
                  rows="3"
                  required
                  value={reviewForm.comment}
                  onChange={(e) =>
                    setReviewForm((p) => ({ ...p, comment: e.target.value }))
                  }
                  className="w-full px-3 py-2 rounded-xl border border-[#3A1E10]/20 bg-white text-xs"
                />
              </div>
              <button
                type="submit"
                className="w-full py-3 rounded-xl bg-[#F39223] text-[#2B1508] font-extrabold text-sm shadow"
              >
                {t.submitReviewBtn}
              </button>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
